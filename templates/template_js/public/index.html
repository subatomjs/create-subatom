import {
  Subatom,
  SubAtomDocs,
  type IRequest,
  type IResponse,
  type ISubatom,
} from "subatom";

const server: ISubatom = new Subatom();

server.setConfig({
  port: 8080,
  host: "localhost",
  maxConcurrentRequests: 1000,
});

// Register base route
server.get("/", (_req: IRequest, res: IResponse) => {
  res.send("Hello from Subatom!");
});

// API Documentation at /docs
SubAtomDocs(server, {
  path: "/docs",
  title: "Subatom Pulse API",
  version: "1.0.0",
});

server.start();
